#include <iostream>

int main() {
    int x;
    int y;
    bool flag;

    x = 10;
    y = 0;
    flag = true;

    for (int i = 0; i < 5; i = i + 1) {
        y = y + i;
    }

    while (x > 0) {
        y = y + x;
        x = x - 1;
    }

    if (flag == true) {
        cout("Result: %d", y);
    } else {
        cout("x is: %d", x);
    }

    return 0;
}
