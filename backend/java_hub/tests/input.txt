class Main {
    public static void main(String[] args) {
print ("enter value: "); 
        Scanner sc = new Scanner(System.in);
        int n;
        n = sc.nextInt();
        int sum = 0;
        
    
        for (int i = 1; i <= n; i++) {
            sum = sum + i;
        }
        print sum;
    }
}
